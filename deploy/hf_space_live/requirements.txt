# opencv is the headless build: the pinned opencv-python pulls GUI libraries
# that cannot load in a container without a display.
# Torch is installed FIRST and from the cu121 index (see Dockerfile), and mmcv's
# compiled _ext is built against that exact build. Re-stating the pins here is
# what stops pip from replacing it: any requirement below that declares a torch
# dependency -- timm did -- leaves the resolver free to pull a newer torch from
# PyPI, and the very next thing that happens is
#     mmcv/_ext...so: undefined symbol: _ZN3c104cuda9SetDeviceEi
# `==2.2.2` is satisfied by the installed `2.2.2+cu121` (PEP 440 local version),
# so this pins without forcing a reinstall from the wrong index.

torchvision==0.17.2

numpy==1.26.4

# 4.45 is the first release with AutoModelForImageTextToText and the Qwen2-VL
# classes, which is what the VLM second opinion needs. It is the SMALLEST bump
# that unblocks it -- Qwen3-VL would need 4.57.
#
# The pin it replaces was load-bearing, so the two build checks below it are the
# safety net rather than an afterthought: `import mmcv._ext` catches the
# `undefined symbol: _ZN3c104cuda9SetDeviceEi` that a dragged-forward torch
# produces, and `register_all_modules()` + importing grounding_dino catches the
# vendored tree failing to load. If either breaks, the BUILD fails and the Space
